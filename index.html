import time class SistemInvestasi: def __init__(self): # Menyimpan data user format: {username: {password: str, saldo: float, persentase_harian: float, riwayat: list}} self.users = {} self.user_aktif = None def menu_utama(self): while True: print("\n" + "="*40) print(" SISTEM SIMULASI INVESTASI HARIAN ") print("="*40) if not self.user_aktif: print("1. Register (Daftar Akun Baru)") print("2. Login Account") print("3. Keluar Aplikasi") print("-"*40) pilihan = input("Pilih menu (1-3): ") if pilihan == "1": self.register() elif pilihan == "2": self.login() elif pilihan == "3": print("\nTerima kasih telah menggunakan sistem kami!") break else: print("[!] Pilihan tidak valid. Silakan coba lagi.") else: print(f"Selamat Datang, {self.user_aktif}!") print(f"Saldo Saat Ini: Rp {self.users[self.user_aktif]['saldo']:,.2f}") print(f"Profit/Hari : {self.users[self.user_aktif]['persentase_harian']}%") print("-"*40) print("1. Deposit (Top Up Saldo)") print("2. Penarikan Dana (Withdrawal)") print("3. Jalankan Simulasi Profit Harian") print("4. Lihat Riwayat Transaksi") print("5. Logout Account") print("-"*40) pilihan = input("Pilih menu (1-5): ") if pilihan == "1": self.deposit() elif pilihan == "2": self.penarikan() elif pilihan == "3": self.simulasi_keuntungan_harian() elif pilihan == "4": self.lihat_riwayat() elif pilihan == "5": self.logout() else: print("[!] Pilihan tidak valid. Silakan coba lagi.") def register(self): print("\n--- MENU REGISTER ---") username = input("Buat Username: ").strip() if not username: print("[!] Username tidak boleh kosong.") return if username in self.users: print("[!] Username sudah terdaftar! Gunakan nama lain.") return password = input("Buat Password: ").strip() if not password: print("[!] Password tidak boleh kosong.") return try: persentase = float(input("Atur Persentase Keuntungan Harian (%): ")) if persentase < 0: print("[!] Persentase keuntungan tidak boleh negatif.") return except ValueError: print("[!] Masukkan angka desimal/bulat yang valid untuk persentase.") return # Daftarkan user baru dengan struktur data lengkap self.users[username] = { "password": password, "saldo": 0.0, "persentase_harian": persentase, "riwayat": ["Akun berhasil dibuat."] } print(f"[✓] Registrasi sukses! Akun '{username}' siap digunakan.") def login(self): print("\n--- MENU LOGIN ---") username = input("Username: ").strip() password = input("Password: ").strip() if username in self.users and self.users[username]["password"] == password: self.user_aktif = username print(f"[✓] Login Berhasil! Selamat datang kembali {username}.") else: print("[!] Username atau Password salah.") def deposit(self): print("\n--- MENU DEPOSIT ---") try: jumlah = float(input("Masukkan nominal Deposit (Rp): ")) if jumlah <= 0: print("[!] Nominal deposit harus lebih dari 0.") return self.users[self.user_aktif]["saldo"] += jumlah pesan_riwayat = f"Deposit masuk: +Rp {jumlah:,.2f}" self.users[self.user_aktif]["riwayat"].append(pesan_riwayat) print(f"[✓] Berhasil deposit Rp {jumlah:,.2f}. Saldo Anda telah diperbarui.") except ValueError: print("[!] Harap masukkan nominal angka yang valid.") def penarikan(self): print("\n--- MENU PENARIKAN (WITHDRAWAL) ---") try: jumlah = float(input("Masukkan nominal Penarikan (Rp): ")) if jumlah <= 0: print("[!] Nominal penarikan harus lebih dari 0.") return saldo_saat_ini = self.users[self.user_aktif]["saldo"] if jumlah > saldo_saat_ini: print(f"[!] Penarikan gagal. Saldo Anda tidak mencukupi (Saldo: Rp {saldo_saat_ini:,.2f}).") return self.users[self.user_aktif]["saldo"] -= jumlah pesan_riwayat = f"Penarikan dana: -Rp {jumlah:,.2f}" self.users[self.user_aktif]["riwayat"].append(pesan_riwayat) print(f"[✓] Berhasil menarik dana Rp {jumlah:,.2f}. Sisa saldo Anda: Rp {self.users[self.user_aktif]['saldo']:,.2f}") except ValueError: print("[!] Harap masukkan nominal angka yang valid.") def simulasi_keuntungan_harian(self): print("\n--- SIMULASI PROFIT HARIAN ---") saldo_sekarang = self.users[self.user_aktif]["saldo"] if saldo_sekarang <= 0: print("[!] Saldo Anda Rp 0. Silakan lakukan deposit terlebih dahulu untuk menghasilkan keuntungan.") return try: hari = int(input("Berapa hari simulasi investasi yang ingin dijalankan?: ")) if hari <= 0: print("[!] Durasi hari harus minimal 1 hari.") return persentase = self.users[self.user_aktif]["persentase_harian"] / 100 print("\nMemproses pertumbuhan investasi harian...") for h in range(1, hari + 1): # Menghitung profit harian berdasarkan sistem bunga majemuk (compound) profit = self.users[self.user_aktif]["saldo"] * persentase self.users[self.user_aktif]["saldo"] += profit print(f" > Hari ke-{h}: Profit +Rp {profit:>12,.2f} | Saldo: Rp {self.users[self.user_aktif]['saldo']:,.2f}") time.sleep(0.1) # Efek jeda simulasi singkat agar interaktif total_keuntungan = self.users[self.user_aktif]["saldo"] - saldo_sekarang pesan_riwayat = f"Simulasi Profit {hari} Hari: +Rp {total_keuntungan:,.2f}" self.users[self.user_aktif]["riwayat"].append(pesan_riwayat) print(f"\n[✓] Simulasi Selesai! Total keuntungan bersih didapat: Rp {total_keuntungan:,.2f}") except ValueError: print("[!] Masukkan jumlah hari yang valid (angka bulat).") def lihat_riwayat(self): print(f"\n--- RIWAYAT TRANSAKSI AKUN: {self.user_aktif} ---") riwayat_list = self.users[self.user_aktif]["riwayat"] for idx, catatan in enumerate(riwayat_list, 1): print(f"{idx}. {catatan}") def logout(self): print(f"\n[✓] Akun '{self.user_aktif}' berhasil logout.") self.user_aktif = None # --- MENJALANKAN PROGRAM --- if __name__ == "__main__": aplikasi = SistemInvestasi() aplikasi.menu_utama()
